# The cross-language contract for ch.csv / bf.csv in this directory.
# E <n=value;...> <wire hex>       : frame encoded from the values. Unlisted channels use their default, else 0. '-' means all defaults
# D <wire hex>  <n=raw/value;...>  : raw and physical values decoded from the frame. A short frame drops overrunning channels
# L <total_bytes> <n:at:bytes;...> : layout

L 13 1:0:4;2:4:2;3:6:1;4:7:2;5:9:4

# Physical values, one per channel.
E 1=1;2=5;3=2;4=-12.3;5=1.5 0100000005000285ffdc050000

# No values: every channel takes its default. CH2 is 0x0001 from the CSV,
# with bit1 cleared by its BF row and bits 0 / 2 left as the default has them.
E - 00000000010000000000000000

# A raw value bypasses lsb and offset.
E 4=0xFF85 0000000001000085ff00000000

# Rounding is half away from zero: 2.5 becomes 3.
E 3=2.5 00000000010003000000000000

# A value past the width of the channel is clamped to it.
E 3=300 000000000100ff000000000000

D 0100000005000285ffdc050000 1=1/1.0;2=5/5.0;3=2/2.0;4=65413/-12.3;5=1500/1.5

# A frame two bytes short of channel 5 drops it rather than zero-filling.
D 0100000005000285ff 1=1/1.0;2=5/5.0;3=2/2.0;4=65413/-12.3
